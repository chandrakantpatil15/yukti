  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

resource "aws_instance" "optimized" {
  ami           = data.aws_instance.target.ami
  instance_type = "t3.medium"  # Downsized from m5.large

}
